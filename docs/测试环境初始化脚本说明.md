#!/bin/bash
# RabbitMQ恢复脚本

set -e

# 配置参数
BACKUP_DIR="/var/backups/rabbitmq/ai-ready-test"
RABBITMQ_ADMIN_USER="admin"
RABBITMQ_ADMIN_PASS=$(sudo grep "^default_pass" /etc/rabbitmq/rabbitmq.config 2>/dev/null | awk -F'"' '{print $2}' || echo "guest")

# 检查备份文件
if [ -z "$1" ]; then
    echo "请指定备份文件"
    echo "可用备份:"
    ls -lt "$BACKUP_DIR/rabbitmq_backup_"*.tar.gz 2>/dev/null || echo "无备份文件"
    exit 1
fi

BACKUP_FILE="$1"

if [ ! -f "$BACKUP_FILE" ]; then
    echo "备份文件不存在: $BACKUP_FILE"
    exit 1
fi

# 确认操作
read -p "这将覆盖RabbitMQ配置，是否继续？(y/N): " -n 1 -r
echo
if [[ ! $REPLY =~ ^[Yy]$ ]]; then
    echo "操作取消"
    exit 1
fi

# 临时目录
TEMP_DIR=$(mktemp -d)

# 解压备份
tar -xzf "$BACKUP_FILE" -C "$TEMP_DIR"

# 停止应用（可选）
# sudo systemctl stop your-application.service

# 清除现有定义
sudo rabbitmqadmin -u "$RABBITMQ_ADMIN_USER" -p "$RABBITMQ_ADMIN_PASS" delete queue name=ai_ready.task_queue
sudo rabbitmqadmin -u "$RABBITMQ_ADMIN_USER" -p "$RABBITMQ_ADMIN_PASS" delete queue name=ai_ready.event_queue
sudo rabbitmqadmin -u "$RABBITMQ_ADMIN_USER" -p "$RABBITMQ_ADMIN_PASS" delete queue name=ai_ready.notification_queue
sudo rabbitmqadmin -u "$RABBITMQ_ADMIN_USER" -p "$RABBITMQ_ADMIN_PASS" delete queue name=ai_ready.dlq

# 恢复定义
sudo rabbitmqadmin -u "$RABBITMQ_ADMIN_USER" -p "$RABBITMQ_ADMIN_PASS" import "$TEMP_DIR/rabbitmq-definitions.json"

# 清理临时目录
rm -rf "$TEMP_DIR"

# 启动应用（可选）
# sudo systemctl start your-application.service

echo "RabbitMQ恢复完成: 配置已从 $BACKUP_FILE 恢复"
```

    echo "  ✗ RabbitMQ服务未运行"

#### 3.2.1 脚本文件：`init-all.sh`

```bash
#!/bin/bash
# 一键初始化脚本

set -e

echo "=== AI-Ready测试环境一键初始化 ==="
echo "开始时间: $(date)"
echo ""

# 定义颜色输出
GREEN='\033[0;32m'
RED='\033[0;31m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

success() {
    echo -e "${GREEN}✓ $1${NC}"
}

error() {
    echo -e "${RED}✗ $1${NC}"
}

warning() {
    echo -e "${YELLOW}⚠ $1${NC}"
}

# 1. 初始化PostgreSQL
echo "1. 初始化PostgreSQL..."
if [ -f "init-postgresql.sh" ]; then
    chmod +x init-postgresql.sh
    if sudo ./init-postgresql.sh; then
        success "PostgreSQL初始化成功"
    else
        error "PostgreSQL初始化失败"
        exit 1
    fi
else
    error "init-postgresql.sh 文件不存在"
    exit 1
fi

echo ""

# 2. 初始化Redis
echo "2. 初始化Redis..."
if [ -f "init-redis.sh" ]; then
    chmod +x init-redis.sh
    if sudo ./init-redis.sh; then
        success "Redis初始化成功"
    else
        error "Redis初始化失败"
        exit 1
    fi
else
    error "init-redis.sh 文件不存在"
    exit 1
fi

echo ""

# 3. 初始化RabbitMQ
echo "3. 初始化RabbitMQ..."
if [ -f "init-rabbitmq.sh" ]; then
    chmod +x init-rabbitmq.sh
    if sudo ./init-rabbitmq.sh; then
        success "RabbitMQ初始化成功"
    else
        error "RabbitMQ初始化失败"
        exit 1
    fi
else
    error "init-rabbitmq.sh 文件不存在"
    exit 1
fi

echo ""

# 4. 创建备份目录
echo "4. 创建备份目录..."
declare -a backup_dirs=(
    "/var/backups/postgresql/ai-ready-test"
    "/var/backups/redis/ai-ready-test"
    "/var/backups/rabbitmq/ai-ready-test"
)

for dir in "${backup_dirs[@]}"; do
    if sudo mkdir -p "$dir"; then
        # 设置正确的权限
        if [[ "$dir" == *"postgresql"* ]]; then
            sudo chown -R postgres:postgres "$dir"
        elif [[ "$dir" == *"redis"* ]]; then
            sudo chown -R redis:redis "$dir"
        elif [[ "$dir" == *"rabbitmq"* ]]; then
            sudo chown -R rabbitmq:rabbitmq "$dir"
        fi
        success "创建目录: $dir"
    else
        error "创建目录失败: $dir"
    fi
done

echo ""

# 5. 验证环境
echo "5. 验证环境初始化..."
if [ -f "check-environment.sh" ]; then
    chmod +x check-environment.sh
    echo "执行环境检查..."
    if ./check-environment.sh; then
        success "环境验证通过"
    else
        warning "环境验证发现一些问题"
    fi
else
    warning "check-environment.sh 文件不存在，跳过环境验证"
fi

echo ""
echo "=== 一键初始化完成 ==="
echo "完成时间: $(date)"
echo ""
echo "重要信息汇总:"
echo "1. PostgreSQL数据库: ai_ready_test"
echo "2. Redis测试实例: 端口 6380"
echo "3. RabbitMQ管理界面: http://localhost:15672"
echo "4. 备份目录已创建在 /var/backups/"

echo "下一步操作:"

echo "2. 配置应用程序连接信息"
echo "3. 运行 check-environment.sh 验证环境状态"

### 3.3 初始化日志记录脚本

#### 3.3.1 脚本文件：`log-initialization.sh`

```bash
#!/bin/bash
# 初始化日志记录脚本

set -e

# 配置参数
LOG_DIR="/var/log/ai-ready-test"
INIT_LOG="$LOG_DIR/initialization.log"
TIMESTAMP=$(date '+%Y-%m-%d %H:%M:%S')

# 创建日志目录
sudo mkdir -p "$LOG_DIR"
sudo chmod 755 "$LOG_DIR"

# 日志函数
log() {
    local level="$1"
    local message="$2"
    echo "[$TIMESTAMP] [$level] $message" | sudo tee -a "$INIT_LOG"
}

log_info() {
    log "INFO" "$1"
}

log_warn() {
    log "WARN" "$1"
}

log_error() {
    log "ERROR" "$1"
}

log_success() {
    log "SUCCESS" "$1"
}

# 开始记录
log_info "=== AI-Ready测试环境初始化开始 ==="
log_info "主机名: $(hostname)"
log_info "操作系统: $(cat /etc/os-release | grep PRETTY_NAME | cut -d= -f2 | tr -d '\"')"
log_info "内核版本: $(uname -r)"

# 记录系统资源
log_info "系统资源:"
log_info "  内存: $(free -h | grep Mem | awk '{print $2}') 总内存"
log_info "  CPU: $(nproc) 核心"

# 记录初始化步骤
record_step() {
    local step_name="$1"
    local command="$2"
    
    log_info "执行步骤: $step_name"
    log_info "命令: $command"
    
    # 执行命令并记录输出
    if eval "$command" 2>&1 | sudo tee -a "$INIT_LOG"; then
        log_success "步骤完成: $step_name"
        return 0
    else
        local exit_code=${PIPESTATUS[0]}
        log_error "步骤失败: $step_name (退出码: $exit_code)"
        return $exit_code
    fi
}

# 示例使用（在实际脚本中调用）
# record_step "初始化PostgreSQL" "./init-postgresql.sh"
# record_step "初始化Redis" "./init-redis.sh"
# record_step "初始化RabbitMQ" "./init-rabbitmq.sh"

# 生成初始化报告
generate_report() {
    local report_file="$LOG_DIR/initialization_report_$(date +%Y%m%d_%H%M%S).txt"
    
    cat > "$report_file" << EOF
AI-Ready测试环境初始化报告
生成时间: $(date)
日志文件: $INIT_LOG

=== 初始化摘要 ===
开始时间: $(grep "初始化开始" "$INIT_LOG" | head -1 | cut -d' ' -f2-)
结束时间: $(date)

=== 服务状态 ===
$(systemctl list-units --type=service | grep -E "(postgresql|redis|rabbitmq)" | sudo tee -a "$report_file")

=== 错误统计 ===
成功步骤: $(grep -c "SUCCESS" "$INIT_LOG")
警告步骤: $(grep -c "WARN" "$INIT_LOG")
错误步骤: $(grep -c "ERROR" "$INIT_LOG")

=== 最后10条日志 ===
$(tail -10 "$INIT_LOG")
EOF
    
    log_info "生成报告: $report_file"
    echo "初始化报告已生成: $report_file"
}

# 清理旧日志
cleanup_old_logs() {
    local retention_days=30
    log_info "清理 $retention_days 天前的日志文件"
    
    find "$LOG_DIR" -name "*.log" -mtime +$retention_days -delete
    find "$LOG_DIR" -name "*report*.txt" -mtime +$retention_days -delete
    
    log_info "日志清理完成"
}

# 结束记录
log_info "=== AI-Ready测试环境初始化结束 ==="

# 生成报告
generate_report

# 清理旧日志
cleanup_old_logs

echo "初始化日志记录完成"
echo "主日志文件: $INIT_LOG"
```

### 4.4 故障排除

#### 常见问题及解决方案：

1. **PostgreSQL连接失败**
   - 检查服务状态：`sudo systemctl status postgresql`
   - 检查端口监听：`sudo netstat -tlnp | grep 5432`
   - 检查防火墙：`sudo ufw status`

2. **Redis连接失败**
   - 检查测试实例：`sudo systemctl status redis-ai-ready-test.service`
   - 检查端口：`redis-cli -p 6380 ping`
   - 验证密码：检查`/etc/redis/ai-ready-test.conf`

3. **RabbitMQ管理界面无法访问**
   - 检查服务：`sudo systemctl status rabbitmq-server`
   - 检查插件：`sudo rabbitmq-plugins list`
   - 检查端口：`sudo netstat -tlnp | grep 15672`

4. **备份失败**
   - 检查目录权限：`ls -la /var/backups/`
   - 检查磁盘空间：`df -h`
   - 检查脚本权限：`ls -la *.sh`
